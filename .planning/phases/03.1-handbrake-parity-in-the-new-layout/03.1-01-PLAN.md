// VideoEncoderOptions.swift
public enum FrameRateMode: String { case constant, peak, variable }            // "Constant Framerate" / "Peak Framerate" / "Variable Framerate"
public enum EncoderSpeed: String { case speed, balanced, quality }              // "Speed" / "Balanced" / "Quality"
public enum VideoProfile: String { case auto, baseline, main, high, main10      // "Auto" "Baseline" "Main" "High" "Main 10"
    public static func options(for codec: VideoCodec) -> [VideoProfile]          // h264 [auto,baseline,main,high]; hevc [auto,main]; hevc10 [auto,main10]; others [auto]
    public func resolved(for codec: VideoCodec) -> VideoProfile }                // self if in options(for:), else .auto
public enum VideoLevel: String { case auto, l30 = "3.0", l31 = "3.1", l32 = "3.2", l40 = "4.0", l41 = "4.1", l42 = "4.2", l50 = "5.0", l51 = "5.1", l52 = "5.2"
    public static func options(for codec: VideoCodec) -> [VideoLevel] }          // h264 → allCases, else [.auto]; displayName auto "Auto", else rawValue
public struct VideoEncoderOptions: Codable, Hashable, Sendable {
    public var frameRateMode: FrameRateMode = .constant
    public var speed: EncoderSpeed = .balanced
    public var profile: VideoProfile = .auto
    public var level: VideoLevel = .auto }

// PictureOptions.swift
public enum CropMode: String { case automatic, conservative, none, custom; public var detects: Bool }   // "Automatic" "Conservative" "None" "Custom"
public enum Anamorphic: String { case none, automatic }                                                 // "None" "Automatic"
public enum PaddingMode: String { case none, letterbox, pillarbox, both }    // "None" "Height (Letterbox)" "Width (Pillarbox)" "Width & Height"
public enum PadColor: String { case black, white, gray, darkGray, custom }   // "Black" "White" "Gray" "Dark Gray" "Custom"
public struct Padding: Codable, Hashable, Sendable {
    public var mode: PaddingMode = .none
    public var color: PadColor = .black
    public var customHex = "000000"                    // RRGGBB, no '#'
    public var isActive: Bool { mode != .none }
    public var red: Double; green: Double; blue: Double (computed, sRGB 0...1) }
public struct PictureOptions: Codable, Hashable, Sendable {
    public var cropMode: CropMode = .custom            // .custom + crop .zero == today's "no crop" (D-05)
    public var allowUpscaling = false
    public var anamorphic: Anamorphic = .none
    public var padding = Padding() }

// VideoFilters.swift (extended in place)
public enum FilterStrength: String { case off, ultralight, light, medium, strong, stronger, verystrong }
public enum DeinterlaceMethod: String { case blend, bob }
public enum DenoiseFilter: String { case nlmeans, hqdn3d }
public enum DenoiseTune: String { case none, film, grain, highMotion = "highmotion", animation, tape, sprite; public var multiplier: Double }
public enum ChromaSmoothTune: String { case none, tiny, small, medium, wide, verywide; public var widthFactor: Double }
public enum SharpenFilter: String { case unsharp, lapsharp }
public enum SharpenTune: String { case none, ultrafine, fine, medium, coarse, verycoarse, film, grain, animation, sprite
    public static func options(for filter: SharpenFilter) -> [SharpenTune]; public func resolved(for filter: SharpenFilter) -> SharpenTune }
public enum ColorspaceTarget: String { case off, bt709, bt601ntsc = "bt601_525", bt601pal = "bt601_625", bt2020 }

                     colorspace(.off) grayscale(false)

public enum AudioSampleRate: Int { case auto = 0, hz8000 = 8000, hz11025 = 11025, hz12000 = 12000, hz16000 = 16000,
    hz22050 = 22050, hz24000 = 24000, hz32000 = 32000, hz44100 = 44100, hz48000 = 48000
    public func resolved(sourceRate: Double, codec: AudioCodec) -> Int }
public enum PassthruCodec: String { case aac, ac3, eac3, dts, dtshd, truehd, mp3, mp2, flac, opus, alac, vorbis
    public static func from(formatID: UInt32) -> PassthruCodec? }
public struct AudioTrackSetting: Codable, Hashable, Identifiable, Sendable {
    public var id: UUID; public var sourceTrackID: Int; public var encoding: AudioEncoding
    public init(id: UUID = UUID(), sourceTrackID: Int, encoding: AudioEncoding) }
public struct AudioOptions: Codable, Hashable, Sendable {
    public var languages: [String] = []
    public var allowedPassthru: Set<PassthruCodec> = Set(PassthruCodec.allCases)
    public var fallback: AudioCodec = .aac
    public var extraFirstTrackOnly = false
    public var sampleRate: AudioSampleRate = .hz48000      // main (behaviour) version; 48 kHz == today (D-05)
    public var gainDb: Double = 0
    public var trackName: String? = nil                    // name of the main (behaviour) version
    public var automaticNaming = false
    public var tracks: [AudioTrackSetting] = [] }           // empty == "use the selection behaviour" (today)
AudioEncoding gains: sampleRate: AudioSampleRate = .hz48000, gainDb: Double = 0, trackName: String? = nil
AudioCodec gains: heAAC ("HE-AAC"), autoPassthru ("Auto Passthru"); order aac, heAAC, alac, passthrough, autoPassthru, none
AudioMixdown gains: surround71 (8 ch, "7.1 Surround")
AudioTrackSelection gains: none ("None"); order none, first, all
MaxResolution gains: uhd4320 = 4320
FrameRate gains: fps5 = 5, fps10 = 10, fps12 = 12, fps15 = 15, fps48 = 48, fps72 = 72, fps75 = 75, fps90 = 90, fps100 = 100, fps120 = 120
VideoCodec gains: proRes422LT ("Apple ProRes 422 LT"), proRes422Proxy ("Apple ProRes 422 Proxy")

// SubtitleOptions.swift
public struct SubtitleTrackSetting: Codable, Hashable, Identifiable, Sendable {
    public var sourceTrackID: Int; public var include = true; public var isDefault = false
    public var forcedOnly = false; public var name: String?; public var id: Int { sourceTrackID } }
public struct SubtitleOptions: Codable, Hashable, Sendable { public var languages: [String] = []; public var tracks: [SubtitleTrackSetting] = [] }
SubtitleTrackInfo gains: name: String? = nil, forcedOnly: Bool? = nil, var isForcedOnly: Bool { forcedOnly ?? false }
AudioTrackInfo gains: name: String? = nil
SubtitleFile gains: offsetMs: Int = 0, title: String? = nil, isDefault = false, isForced = false

// ChapterOptions.swift / MetadataTags.swift / LanguageFilter.swift
public struct ChapterOptions: Codable, Hashable, Sendable {
    public var markers: Bool? = nil                        // nil = follow keepMetadata (today's behaviour)
    public var titles: [String] = []                       // per source chapter index; "" = keep source title
    public func writesMarkers(keepMetadata: Bool) -> Bool { markers ?? keepMetadata } }
public struct MetadataTags: Codable, Hashable, Sendable {
    public var title, actors, director, releaseDate, comment, genre, description, plot: String?   // nil = not set
    public var isEmpty: Bool }                             // true when every field is nil or whitespace
public enum LanguageFilter {
    public static func normalized(_ tag: String) -> String?            // ISO 639-2 alpha-3, lowercased; "", "any", "und" → nil
    public static func matches(_ language: String?, in list: [String]) -> Bool }   // empty/all-nil list → true

// EncodeSettings new stored properties
